BASE URL par défaut :
http://localhost:8017

Si API_TOKEN est défini :
export TOKEN="ton_token"
alias apicurl='curl -sS -H "Authorization: Bearer '$TOKEN'"'

Sans auth :
alias apicurl='curl -sS'

1) Santé
apicurl "http://localhost:8017/health"

2) Recherche série One Piece
apicurl --get "http://localhost:8017/search" \
  --data-urlencode "q=one piece" \
  --data-urlencode "kind=series" \
  --data-urlencode "mode=all" \
  --data-urlencode "limit=10"

Attendus minimaux :
- `data[0].title` cohérent
- `data[0].score` élevé
- `data[0].title_vo` si disponible
- `data[0].vf` et `data[0].vo` sur un bon résultat série enrichi

3) Resolve série One Piece
apicurl --get "http://localhost:8017/search/resolve" \
  --data-urlencode "q=one piece" \
  --data-urlencode "kind=series" \
  --data-urlencode "limit=10"

Attendus minimaux :
- `data.best.slug = One-piece-Edition-originale`
- `data.confidence` non vide

4) Fiche série
apicurl "http://localhost:8017/series/One-piece-Edition-originale"

Attendus minimaux :
- `data.title = One Piece`
- `data.vf.volumes` non nul
- `data.vo.volumes` non nul

5) Recherche volume One Piece tome 91
apicurl --get "http://localhost:8017/search/resolve" \
  --data-urlencode "q=one piece tome 91" \
  --data-urlencode "kind=volume" \
  --data-urlencode "limit=10"

Attendus minimaux :
- `data.best.series_slug = One-Piece`
- `data.best.volume_slug = vol-91`
- `data.best.number = 91` si l'enrichissement l'a remonté

6) Fiche volume 91
apicurl "http://localhost:8017/volume/One-Piece/vol-91"

Attendus minimaux :
- `data.number = 91`
- `data.number_int = 91`
- `data.vf.volumes` non nul
- `data.vo.volumes` non nul

7) News globales
apicurl "http://localhost:8017/news/global?limit=10"

8) News série
apicurl "http://localhost:8017/news/series/One-piece-Edition-originale?limit=10"

9) Planning VF avril 2026
apicurl --get "http://localhost:8017/planning" \
  --data-urlencode "section=manga-vf" \
  --data-urlencode "year=2026" \
  --data-urlencode "month=4" \
  --data-urlencode "publisher=Glénat"

10) ETag
curl -i "http://localhost:8017/series/One-piece-Edition-originale"
curl -i "http://localhost:8017/series/One-piece-Edition-originale" -H 'If-None-Match: "<etag_précédent>"'

11) Batch de smoke tests
python scripts/run_api_smoke_tests.py --base-url "http://localhost:8017" --token "$TOKEN"
